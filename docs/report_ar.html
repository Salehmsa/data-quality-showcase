سكربت التنظيف المقترح

import pandas as pd

df = pd.read_csv("messy_sales.csv")

# 1) clean column names
df.columns = df.columns.str.strip()

# 2) normalise text categories (strip + lower; then map variants)
df["City"] = df["City"].astype(str).str.strip().str.lower()

df["Unit Price"] = pd.to_numeric(df["Unit Price"].astype(str).str.replace(r"[^\d.\-]", "", regex=True), errors="coerce")
df["order_date"] = pd.to_datetime(df["order_date"], errors="coerce")

# exact duplicates
before = len(df)
df = df.drop_duplicates()
print("removed duplicates:", before - len(df))

# constant columns carry no information
df = df.drop(columns=['Region', 'Notes'])

# ---- needs a human decision (do not automate blindly) ----
# Quantity: 16 negative values -> returns or entry errors? 
# df = df[df["Quantity"] >= 0]
# Discount_rate: 8 values > 1 in a rate column
# df = df[df["Discount_rate"] <= 1]
# Delivery Days: placeholder 999 (21x)
# df["Delivery Days"] = df["Delivery Days"].replace(999, pd.NA)
# Notes: 71.1% missing -> choose: drop column / impute (median, mode, "Unknown")

df.to_csv("cleaned.csv", index=False)
print(df.shape)
